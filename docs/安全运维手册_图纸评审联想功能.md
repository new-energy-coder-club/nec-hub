```typescript
// 多因素认证配置
const MFA_CONFIG = {
  enabled: true,
  methods: ['password', 'sms', 'totp'],
  required: ['password', 'sms'], // 必须同时使用密码和短信
  backupCodes: 10,
  sessionTimeout: 3600 // 1小时
};

// 认证服务
class AuthenticationService {
  async authenticate(credentials: Credentials): Promise<AuthResult> {
    // 第一步: 密码验证
    const passwordValid = await this.verifyPassword(credentials.password);
    if (!passwordValid) {
      throw new Error('密码验证失败');
    }
    
    // 第二步: 短信验证
    const smsValid = await this.verifySMS(credentials.smsCode);
    if (!smsValid) {
      throw new Error('短信验证失败');
    }
    
    // 生成访问令牌
    const token = await this.generateToken(credentials.userId);
    
    return {
      success: true,
      token,
      expiresIn: MFA_CONFIG.sessionTimeout
    };
  }
}
```

### 2. 授权控制

**基于角色的访问控制 (RBAC)**:
```typescript
// 角色定义
enum Role {
  USER = 'user',
  DESIGNER = 'designer',
  PROJECT_MANAGER = 'project_manager',
  ADMIN = 'admin',
  SUPER_ADMIN = 'super_admin'
}

// 权限定义
enum Permission {
  // 评审单权限
  REVIEW_CREATE = 'review:create',
  REVIEW_READ = 'review:read',
  REVIEW_UPDATE = 'review:update',
  REVIEW_DELETE = 'review:delete',
  
  // 联想功能权限
  SUGGESTION_READ = 'suggestion:read',
  SUGGESTION_WRITE = 'suggestion:write',
  
  // 系统管理权限
  SYSTEM_CONFIG = 'system:config',
  USER_MANAGE = 'user:manage',
  AUDIT_VIEW = 'audit:view'
}

// 角色权限映射
const ROLE_PERMISSIONS: Record<Role, Permission[]> = {
  [Role.USER]: [
    Permission.REVIEW_CREATE,
    Permission.REVIEW_READ,
    Permission.SUGGESTION_READ
  ],
  [Role.DESIGNER]: [
    Permission.REVIEW_CREATE,
    Permission.REVIEW_READ,
    Permission.REVIEW_UPDATE,
    Permission.SUGGESTION_READ,
    Permission.SUGGESTION_WRITE
  ],
  [Role.PROJECT_MANAGER]: [
    Permission.REVIEW_CREATE,
    Permission.REVIEW_READ,
    Permission.REVIEW_UPDATE,
    Permission.REVIEW_DELETE,
    Permission.SUGGESTION_READ,
    Permission.SUGGESTION_WRITE
  ],
  [Role.ADMIN]: [
    Permission.REVIEW_CREATE,
    Permission.REVIEW_READ,
    Permission.REVIEW_UPDATE,
    Permission.REVIEW_DELETE,
    Permission.SUGGESTION_READ,
    Permission.SUGGESTION_WRITE,
    Permission.SYSTEM_CONFIG,
    Permission.USER_MANAGE
  ],
  [Role.SUPER_ADMIN]: Object.values(Permission)
};

// 授权服务
class AuthorizationService {
  async checkPermission(
    userId: string, 
    resource: string, 
    action: string
  ): Promise<boolean> {
    const user = await this.getUser(userId);
    const userRoles = user.roles;
    
    // 检查用户角色权限
    for (const role of userRoles) {
      const permissions = ROLE_PERMISSIONS[role];
      const requiredPermission = `${resource}:${action}`;
      
      if (permissions.includes(requiredPermission as Permission)) {
        return true;
      }
    }
    
    return false;
  }
  
  async getUserPermissions(userId: string): Promise<Permission[]> {
    const user = await this.getUser(userId);
    const permissions: Permission[] = [];
    
    for (const role of user.roles) {
      permissions.push(...ROLE_PERMISSIONS[role]);
    }
    
    return [...new Set(permissions)]; // 去重
  }
}
```

```typescript
// 数据加密服务
class DataEncryptionService {
  private readonly algorithm = 'aes-256-gcm';
  private readonly keyLength = 32;
  private readonly ivLength = 16;
  private readonly tagLength = 16;
  
  async encrypt(data: string, key: string): Promise<string> {
    const iv = crypto.randomBytes(this.ivLength);
    const cipher = crypto.createCipher(this.algorithm, key);
    cipher.setAAD(Buffer.from('nec-hub', 'utf8'));
    
    let encrypted = cipher.update(data, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    
    const tag = cipher.getAuthTag();
    
    return iv.toString('hex') + ':' + tag.toString('hex') + ':' + encrypted;
  }
  
  async decrypt(encryptedData: string, key: string): Promise<string> {
    const parts = encryptedData.split(':');
    const iv = Buffer.from(parts[0], 'hex');
    const tag = Buffer.from(parts[1], 'hex');
    const encrypted = parts[2];
    
    const decipher = crypto.createDecipher(this.algorithm, key);
    decipher.setAAD(Buffer.from('nec-hub', 'utf8'));
    decipher.setAuthTag(tag);
    
    let decrypted = decipher.update(encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    
    return decrypted;
  }
}
```

```typescript
class DataMaskingService {
  // 项目编号脱敏
  static maskProjectId(projectId: string): string {
    if (projectId.length <= 6) return projectId;
    return projectId.replace(/(.{3}).*(.{3})/, '$1***$2');
  }
  
  // 机型名称脱敏
  static maskMachineName(machineName: string): string {
    if (machineName.length <= 4) return machineName;
    return machineName.replace(/(.{2}).*(.{2})/, '$1***$2');

  static maskUserInfo(userInfo: any): any {
    return {
      ...userInfo,
```

### 1. 输入验证

**输入验证服务**:
```typescript
// 输入验证器
class InputValidator {
  // SQL 注入防护
  static validateSQLInput(input: string): boolean {
    const sqlPatterns = [
      /(\b(SELECT|INSERT|UPDATE|DELETE|DROP|CREATE|ALTER|EXEC|UNION|SCRIPT)\b)/i,
      /(\b(OR|AND)\s+\d+\s*=\s*\d+)/i,
      /(--|\/\*|\*\/|;|'|"|`)/,
      /(\b(WAITFOR|DELAY)\b)/i
    ];
    
    return !sqlPatterns.some(pattern => pattern.test(input));
  }
  
  // XSS 防护
  static validateXSSInput(input: string): boolean {
    const xssPatterns = [
      /<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi,
      /<iframe\b[^<]*(?:(?!<\/iframe>)<[^<]*)*<\/iframe>/gi,
      /javascript:/gi,
      /on\w+\s*=/gi,
      /<img\b[^>]*src\s*=\s*["']?javascript:/gi
    ];
    
    return !xssPatterns.some(pattern => pattern.test(input));
  }
  
  // 文件上传验证
  static validateFileUpload(file: File): boolean {
    const allowedTypes = ['.pdf', '.dwg', '.step', '.zip'];
    const maxSize = 10 * 1024 * 1024; // 10MB
    
    // 检查文件类型
    const fileExtension = file.name.toLowerCase().substring(file.name.lastIndexOf('.'));
    if (!allowedTypes.includes(fileExtension)) {
      return false;
    }
    
    // 检查文件大小
    if (file.size > maxSize) {
      return false;
    }
    
    return true;
  }
}
```

### 2. 访问控制

**API 访问控制**:
```typescript
// 访问控制中间件
class AccessControlMiddleware {
  static async checkAccess(req: Request, res: Response, next: NextFunction): Promise<void> {
    const clientIP = this.getClientIP(req);
    const userAgent = req.get('User-Agent') || '';
    const timestamp = Date.now();
    
    // IP 白名单检查
    if (!this.isIPWhitelisted(clientIP)) {
      await this.logAccessDenied(clientIP, 'IP_NOT_WHITELISTED', req.url);
      return res.status(403).json({ error: '访问被拒绝' });
    }
    
    // 请求频率限制
    if (await this.isRateLimited(clientIP)) {
      await this.logAccessDenied(clientIP, 'RATE_LIMIT_EXCEEDED', req.url);
      return res.status(429).json({ error: '请求过于频繁' });
    }
    
    // 用户代理检查
    if (!this.isValidUserAgent(userAgent)) {
      await this.logAccessDenied(clientIP, 'INVALID_USER_AGENT', req.url);
      return res.status(403).json({ error: '无效的用户代理' });
    }
    
    // 记录访问日志
    await this.logAccess(clientIP, userAgent, req.url, timestamp);
    
    next();
  }
  
  private static isIPWhitelisted(ip: string): boolean {
    const whitelist = process.env.IP_WHITELIST?.split(',') || [];
    return whitelist.includes(ip) || whitelist.includes('*');
  }
  
  private static async isRateLimited(ip: string): Promise<boolean> {
    const key = `rate_limit_${ip}`;
    const count = await this.getRequestCount(key);
    const maxRequests = parseInt(process.env.MAX_REQUESTS_PER_HOUR || '1000');
    return count > maxRequests;
  }
  
  private static isValidUserAgent(userAgent: string): boolean {
    // 检查是否为有效的浏览器或 API 客户端
    const validPatterns = [
      /Mozilla\/\d+\.\d+/,
      /PostmanRuntime/,
      /curl\/\d+\.\d+/,
      /NEC-Hub-Client/
    ];
    
    return validPatterns.some(pattern => pattern.test(userAgent));
  }
}
```

```typescript
class AuditLogger {
  private static readonly AUDIT_LOG_KEY = 'audit_logs';
  
  static async logEvent(event: AuditEvent): Promise<void> {
    const logEntry = {
      id: this.generateLogId(),
      timestamp: new Date().toISOString(),
      userId: event.userId,
      action: event.action,
      resource: event.resource,
      details: event.details,
      ip: event.ip,
      userAgent: event.userAgent,
      result: event.result,
      riskLevel: this.calculateRiskLevel(event)
    };
    
    // 存储审计日志
    await this.storeLog(logEntry);
    
    // 检查高风险事件
    if (logEntry.riskLevel === 'HIGH') {
      await this.triggerSecurityAlert(logEntry);
    }
  }
  
  private static calculateRiskLevel(event: AuditEvent): 'LOW' | 'MEDIUM' | 'HIGH' {
    const highRiskActions = ['DELETE', 'CONFIG_CHANGE', 'PERMISSION_GRANT'];
    const mediumRiskActions = ['UPDATE', 'CREATE', 'LOGIN'];
    
    if (highRiskActions.includes(event.action)) {
      return 'HIGH';
    } else if (mediumRiskActions.includes(event.action)) {
      return 'MEDIUM';
    } else {
      return 'LOW';
    }
  }
  
  private static async storeLog(logEntry: AuditLogEntry): Promise<void> {
    // 存储到安全日志系统
    console.log('AUDIT_LOG:', JSON.stringify(logEntry));
  }
  
  private static async triggerSecurityAlert(logEntry: AuditLogEntry): Promise<void> {
    // 触发安全告警
    console.log('SECURITY_ALERT:', JSON.stringify(logEntry));
  }
}
```

```typescript
// 安全监控器
class SecurityMonitor {
  private static readonly MONITORING_INTERVAL = 60000; // 1分钟
  
  static async startMonitoring(): Promise<void> {
    setInterval(async () => {
      await this.performSecurityChecks();
    }, this.MONITORING_INTERVAL);
  }
  
  private static async performSecurityChecks(): Promise<void> {
    // 检查异常登录
    await this.checkAnomalousLogins();
    
    // 检查权限提升
    await this.checkPrivilegeEscalation();
    
    // 检查数据泄露
    await this.checkDataLeakage();
    
    // 检查系统入侵
    await this.checkSystemIntrusion();
  }
  
  private static async checkAnomalousLogins(): Promise<void> {
    // 检查异常登录模式
    const recentLogins = await this.getRecentLogins();
    
    for (const login of recentLogins) {
      if (await this.isAnomalousLogin(login)) {
        await this.triggerSecurityAlert('ANOMALOUS_LOGIN', login);
      }
    }
  }
  
  private static async isAnomalousLogin(login: LoginRecord): Promise<boolean> {
    // 检查登录地点
    const isNewLocation = await this.isNewLocation(login.ip, login.userId);
    
    // 检查登录时间
    const isUnusualTime = this.isUnusualTime(login.timestamp);
    
    // 检查登录频率
    const isHighFrequency = await this.isHighFrequencyLogin(login.userId);
    
    return isNewLocation || isUnusualTime || isHighFrequency;
  }
}
```

**安全事件分类**:

- **级别 2**: 严重安全事件 (如：权限提升攻击)
- **级别 3**: 紧急安全事件 (如：数据泄露)

**应急响应流程**:
```typescript
// 安全事件响应服务
class SecurityIncidentResponse {
  async handleIncident(incident: SecurityIncident): Promise<void> {
    // 1. 事件分类
    const severity = this.classifyIncident(incident);
    
    // 2. 启动响应流程
    switch (severity) {
      case 'LEVEL_1':
        await this.handleLevel1Incident(incident);
        break;
      case 'LEVEL_2':
        await this.handleLevel2Incident(incident);
        break;
      case 'LEVEL_3':
        await this.handleLevel3Incident(incident);
        break;
    }
    
    // 3. 记录事件
    await this.logIncident(incident, severity);
    
    // 4. 通知相关人员
    await this.notifyStakeholders(incident, severity);
  }
  
  private async handleLevel1Incident(incident: SecurityIncident): Promise<void> {
    // 记录事件
    console.log('Level 1 安全事件:', incident);
    
    // 自动处理
    await this.autoHandleIncident(incident);
  }
  
  private async handleLevel2Incident(incident: SecurityIncident): Promise<void> {
    // 立即通知安全团队
    await this.notifySecurityTeam(incident);
    
    // 隔离受影响系统
    await this.isolateAffectedSystems(incident);
    
    // 启动调查
    await this.startInvestigation(incident);
  }
  
  private async handleLevel3Incident(incident: SecurityIncident): Promise<void> {
    // 立即通知管理层
    await this.notifyManagement(incident);
    
    // 启动紧急响应
    await this.activateEmergencyResponse(incident);
    
    // 联系外部安全专家
    await this.contactExternalExperts(incident);
  }
}
```

```typescript
class DisasterRecoveryService {
  private static readonly BACKUP_RETENTION_DAYS = 30;
```
