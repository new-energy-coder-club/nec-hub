```javascript
// 防抖处理
let suggestionTimeout;

// 项目编号输入处理
function handleProjectInput(input) {
  clearTimeout(suggestionTimeout);
  const query = input.value.trim();
  
  if (query.length < 2) {
    hideProjectSuggestions();
    return;
  }
  
  suggestionTimeout = setTimeout(() => {
    fetchSuggestions(query, 'project', input);
  }, 300);
}

// 获取联想建议
async function fetchSuggestions(query, type, input) {
  try {
    const projectId = type === 'machine' ? document.getElementById('dr-form').projectId.value : undefined;
    const url = `/api/drawing-review/suggestions?q=${encodeURIComponent(query)}&type=${type}${projectId ? '&projectId=' + encodeURIComponent(projectId) : ''}`;
    
    const data = await apiDr(url);
    if (data.ok && data.suggestions) {
      currentSuggestions[type] = data.suggestions;
      showSuggestions(type, input, data.suggestions);
    }
  } catch (err) {
    console.error("获取联想建议失败:", err);
  }
}

// 显示联想建议
function showSuggestions(type, input, suggestions) {
  const container = document.getElementById(`${type}-suggestions`);
  if (!container) return;
  
  if (suggestions.length === 0) {
    container.classList.add('hidden');
    return;
  }
  
  container.innerHTML = suggestions.map(s => `
    <div class="px-3 py-2 hover:bg-gray-100 cursor-pointer border-b border-gray-100 last:border-b-0"
         onclick="selectSuggestion('${type}', '${s.value}')">
      <div class="text-sm font-medium text-gray-800">${s.label}</div>
      ${s.count ? `<div class="text-xs text-gray-500">使用 ${s.count} 次</div>` : ''}
    </div>
  `).join('');
  
  container.classList.remove('hidden');
}
```

## 📊 性能优化

### 1. 缓存策略

**客户端缓存**:
```javascript
// 缓存联想结果
const suggestionCache = new Map();

async function fetchSuggestions(query, type, input) {
  const cacheKey = `${type}:${query}`;
  
  if (suggestionCache.has(cacheKey)) {
    const cached = suggestionCache.get(cacheKey);
    if (Date.now() - cached.timestamp < 300000) { // 5分钟缓存
      showSuggestions(type, input, cached.data);
      return;
    }
  }
  
  // 发起新的 API 请求
  const data = await apiDr(url);
  suggestionCache.set(cacheKey, {
    data: data.suggestions,
    timestamp: Date.now()
  });
  
  showSuggestions(type, input, data.suggestions);
}
```

**服务端缓存**:
```typescript
// 使用 Cloudflare Workers KV 缓存
class SuggestionService {
  private async getCachedSuggestions(key: string): Promise<SuggestionResult[] | null> {
    try {
      const cached = await this.env.REGISTRY.get(`suggestions:${key}`);
      return cached ? JSON.parse(cached) : null;
    } catch {
      return null;
    }
  }
  
  private async setCachedSuggestions(key: string, suggestions: SuggestionResult[]): Promise<void> {
    try {
      await this.env.REGISTRY.put(`suggestions:${key}`, JSON.stringify(suggestions), {
        expirationTtl: 300 // 5分钟过期
      });
    } catch (error) {
      console.warn("缓存联想结果失败:", error);
    }
  }
}
```

### 2. 防抖优化

**前端防抖**:
```javascript
// 高级防抖实现
function debounce(func, wait, immediate) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      timeout = null;
      if (!immediate) func(...args);
    };
    const callNow = immediate && !timeout;
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
    if (callNow) func(...args);
  };
}

// 使用防抖
const debouncedFetchSuggestions = debounce(fetchSuggestions, 300);
```

```typescript
function validateSuggestionInput(query: string, type: string): boolean {
  // 验证查询字符串
  if (!query || query.length < 2) return false;
  if (query.length > 100) return false;
  
  // 验证类型参数
  const validTypes = ['project', 'machine', 'all'];
  if (!validTypes.includes(type)) return false;
  
  // 防止 SQL 注入
  const sanitizedQuery = query.replace(/[<>'"%;()&+]/g, '');
  return sanitizedQuery === query;
}
```

```typescript
    }));
  }
```
```typescript
}
```
