```typescript
// 监控指标收集
class MetricsCollector {
  private metrics: Map<string, number> = new Map();
  
  // 记录 API 响应时间
  recordApiResponseTime(endpoint: string, duration: number): void {
    const key = `api_response_time_${endpoint}`;
    this.metrics.set(key, duration);
    
    // 上报到监控系统
    this.reportMetric(key, duration);
  }
  
  // 记录错误
  recordError(endpoint: string, error: Error): void {
    const key = `api_error_${endpoint}`;
    const count = this.metrics.get(key) || 0;
    this.metrics.set(key, count + 1);
    
    // 上报到监控系统
    this.reportMetric(key, count + 1);
  }
  
  // 记录业务指标
  recordBusinessMetric(metric: string, value: number): void {
    this.metrics.set(metric, value);
    this.reportMetric(metric, value);
  }
  
  private reportMetric(name: string, value: number): void {
    // 发送到监控系统
    console.log(`METRIC: ${name}=${value}`);
  }
}
```

// 日志级别配置

  static info(message: string, meta?: any): void {
    console.log(JSON.stringify({
      level: "info",
      message,
      meta,
      timestamp: new Date().toISOString()
    }));
  }
  
  static error(message: string, error?: Error, meta?: any): void {
    console.error(JSON.stringify({
      level: "error",
      message,
      error: error?.message,
      stack: error?.stack,
      meta,
      timestamp: new Date().toISOString()
    }));
  }
  
  static warn(message: string, meta?: any): void {
    console.warn(JSON.stringify({
      level: "warn",
      message,
      meta,
      timestamp: new Date().toISOString()
    }));
  }
}

### 1. 自动化监控

**健康检查脚本**:
```bash
#!/bin/bash
# 系统健康检查脚本

HEALTH_URL="https://your-domain.com/api/health"
LOG_FILE="/var/log/health-check.log"
ALERT_EMAIL="admin@newenergycoder.club"

# 执行健康检查
response=$(curl -s -o /dev/null -w "%{http_code}" $HEALTH_URL)
timestamp=$(date '+%Y-%m-%d %H:%M:%S')

if [ $response -eq 200 ]; then
  echo "$timestamp - ✅ 系统正常" >> $LOG_FILE
else
  echo "$timestamp - ❌ 系统异常 (HTTP $response)" >> $LOG_FILE
  
  # 发送告警
  echo "系统异常告警: HTTP $response" | mail -s "NEC Hub 系统异常" $ALERT_EMAIL
fi
```

**性能监控脚本**:
```bash
#!/bin/bash
# 性能监控脚本

API_URL="https://your-domain.com/api/drawing-review/suggestions?q=test"
LOG_FILE="/var/log/performance.log"

# 测量响应时间
start_time=$(date +%s%N)
response=$(curl -s -o /dev/null -w "%{http_code}" $API_URL)
end_time=$(date +%s%N)

response_time=$(( (end_time - start_time) / 1000000 ))
timestamp=$(date '+%Y-%m-%d %H:%M:%S')

echo "$timestamp - 响应时间: ${response_time}ms, 状态码: $response" >> $LOG_FILE

# 检查是否超过阈值
if [ $response_time -gt 1000 ]; then
  echo "$timestamp - ⚠️ 响应时间过长: ${response_time}ms" >> $LOG_FILE
  echo "性能告警: 响应时间 ${response_time}ms" | mail -s "NEC Hub 性能告警" admin@newenergycoder.club
fi
```

**自动扩缩容**:
```typescript
// 自动扩缩容服务
class AutoScalingService {
  private readonly CPU_THRESHOLD_UP = 80;
  private readonly CPU_THRESHOLD_DOWN = 20;
  private readonly MIN_INSTANCES = 1;
  private readonly MAX_INSTANCES = 10;
  
  async checkAndScale(): Promise<void> {
    const metrics = await this.getSystemMetrics();
    
    if (metrics.cpuUsage > this.CPU_THRESHOLD_UP) {
      await this.scaleUp();
    } else if (metrics.cpuUsage < this.CPU_THRESHOLD_DOWN) {
      await this.scaleDown();
    }
  }
  
  private async scaleUp(): Promise<void> {
    const currentInstances = await this.getCurrentInstances();
    if (currentInstances < this.MAX_INSTANCES) {
      await this.addInstance();
      console.log(`扩容完成: ${currentInstances} -> ${currentInstances + 1}`);
    }
  }
  
  private async scaleDown(): Promise<void> {
    const currentInstances = await this.getCurrentInstances();
    if (currentInstances > this.MIN_INSTANCES) {
      await this.removeInstance();
      console.log(`缩容完成: ${currentInstances} -> ${currentInstances - 1}`);
    }
  }
}
```

## 🔒 安全监控

### 1. 安全事件监控

**安全事件类型**:
- **异常访问**: 大量请求来自同一 IP
- **权限异常**: 越权访问尝试
- **数据异常**: 异常数据查询或修改
- **系统异常**: 系统配置被修改

**安全监控配置**:
```typescript
// 安全监控服务
class SecurityMonitoringService {
  private suspiciousIPs: Set<string> = new Set();
  private readonly MAX_REQUESTS_PER_IP = 1000; // 每小时
  
  async monitorRequest(request: Request): Promise<boolean> {
    const clientIP = this.getClientIP(request);
    const timestamp = Date.now();
    
    // 检查 IP 请求频率
    if (await this.isRateLimited(clientIP)) {
      await this.logSecurityEvent('RATE_LIMIT_EXCEEDED', { ip: clientIP });
      return false;
    }
    
    // 检查异常访问模式
    if (await this.isSuspiciousAccess(request)) {
      await this.logSecurityEvent('SUSPICIOUS_ACCESS', { ip: clientIP, url: request.url });
      return false;
    }
    
    return true;
  }
  
  private async isRateLimited(ip: string): Promise<boolean> {
    const key = `rate_limit_${ip}`;
    const count = await this.getRequestCount(key);
    return count > this.MAX_REQUESTS_PER_IP;
  }
  
  private async logSecurityEvent(type: string, data: any): Promise<void> {
    const event = {
      type,
      data,
      timestamp: new Date().toISOString(),
      severity: this.getEventSeverity(type)
    };
    
    // 记录安全事件
    console.log('SECURITY_EVENT:', JSON.stringify(event));
    
    // 发送安全告警
    await this.sendSecurityAlert(event);
  }
}
```

class AuditLogService {
