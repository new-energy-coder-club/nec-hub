**插件管理器**:
```typescript
// src/drawing-review/plugins/plugin.manager.ts
import { SuggestionPlugin, PluginConfig } from './base.plugin';

export class PluginManager {
  private plugins: Map<string, SuggestionPlugin> = new Map();
  private configs: Map<string, PluginConfig> = new Map();
  
  // 注册插件
  async registerPlugin(plugin: SuggestionPlugin, config: PluginConfig): Promise<void> {
    try {
      await plugin.initialize(config);
      this.plugins.set(plugin.name, plugin);
      this.configs.set(plugin.name, config);
      console.log(`插件 ${plugin.name} 注册成功`);
    } catch (error) {
      console.error(`插件 ${plugin.name} 注册失败:`, error);
      throw error;
    }
  }
  
  // 执行所有插件
  async executePlugins(context: SuggestionContext, query: string): Promise<SuggestionResult[]> {
    const results: SuggestionResult[] = [];
    const sortedPlugins = this.getSortedPlugins();
    
    for (const plugin of sortedPlugins) {
      try {
        const config = this.configs.get(plugin.name);
        if (!config?.enabled) continue;
        
        const pluginResults = await this.executeWithTimeout(plugin, context, query, config.timeout);
        results.push(...pluginResults);
      } catch (error) {
        console.error(`插件 ${plugin.name} 执行失败:`, error);
        // 继续执行其他插件
      }
    }
    
    return results;
  }
  
  // 获取排序后的插件
  private getSortedPlugins(): SuggestionPlugin[] {
    return Array.from(this.plugins.values())
      .sort((a, b) => {
        const configA = this.configs.get(a.name);
        const configB = this.configs.get(b.name);
        return (configB?.priority || 0) - (configA?.priority || 0);
      });
  }
  
  // 带超时的插件执行
  private async executeWithTimeout(
    plugin: SuggestionPlugin, 
    context: SuggestionContext, 
    query: string, 
    timeout: number
  ): Promise<SuggestionResult[]> {
    return Promise.race([
      plugin.execute(context, query),
      new Promise<SuggestionResult[]>((_, reject) => 
        setTimeout(() => reject(new Error('插件执行超时')), timeout)
      )
    ]);
  }
  
  // 卸载插件
  async unregisterPlugin(name: string): Promise<void> {
    const plugin = this.plugins.get(name);
    if (plugin) {
      await plugin.cleanup();
      this.plugins.delete(name);
      this.configs.delete(name);
      console.log(`插件 ${name} 卸载成功`);
    }
  }
  
  // 获取插件列表
  getPluginList(): PluginInfo[] {
    return Array.from(this.plugins.values()).map(plugin => plugin.getInfo());
  }
}
```

```typescript
// src/drawing-review/datasources/erp.datasource.ts
import { DataSource, QueryOptions } from './base.datasource';

export class ERPDataSource implements DataSource {
  name = "erp-system";
  type = "external-api";
  description = "ERP 系统集成";
  
  private apiUrl: string;
  private apiKey: string;
  private connected: boolean = false;
  
  constructor(apiUrl: string, apiKey: string) {
    this.apiUrl = apiUrl;
    this.apiKey = apiKey;
  }
  
  async connect(): Promise<void> {
    try {
      // 测试连接
      const response = await fetch(`${this.apiUrl}/health`, {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json'
        }
      });
      
      if (response.ok) {
        this.connected = true;
        console.log('ERP 数据源连接成功');
      } else {
        throw new Error(`ERP 连接失败: ${response.status}`);
      }
    } catch (error) {
      console.error('ERP 数据源连接失败:', error);
      throw error;
    }
  }
  
  async query(query: string, options?: QueryOptions): Promise<any[]> {
    if (!this.connected) {
      throw new Error('ERP 数据源未连接');
    }
    
    try {
      const params = new URLSearchParams({
        q: query,
        limit: options?.limit?.toString() || '10',
        offset: options?.offset?.toString() || '0'
      });
      
      const response = await fetch(`${this.apiUrl}/materials?${params}`, {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json'
        }
      });
      
      if (!response.ok) {
        throw new Error(`ERP 查询失败: ${response.status}`);
      }
      
      const data = await response.json();
      return this.transformERPData(data);
    } catch (error) {
      console.error('ERP 查询失败:', error);
      throw error;
    }
  }
  
  async disconnect(): Promise<void> {
    this.connected = false;
    console.log('ERP 数据源已断开');
  }
  
  async healthCheck(): Promise<boolean> {
    try {
      const response = await fetch(`${this.apiUrl}/health`, {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`
        }
      });
      return response.ok;
    } catch {
      return false;
    }
  }
  
  private transformERPData(data: any[]): any[] {
    // 转换 ERP 数据格式
    return data.map(item => ({
      code: item.materialCode,
      name: item.materialName,
      specification: item.specification,
      unit: item.unit,
      category: item.category,
      supplier: item.supplier,
      price: item.price,
      stock: item.stock
    }));
  }
}
```

```typescript
// src/drawing-review/datasources/datasource.manager.ts
import { DataSource } from './base.datasource';

export class DataSourceManager {
  private dataSources: Map<string, DataSource> = new Map();
  private activeConnections: Set<string> = new Set();
  
  // 注册数据源
  async registerDataSource(dataSource: DataSource): Promise<void> {
    try {
      await dataSource.connect();
      this.dataSources.set(dataSource.name, dataSource);
      this.activeConnections.add(dataSource.name);
      console.log(`数据源 ${dataSource.name} 注册成功`);
    } catch (error) {
      console.error(`数据源 ${dataSource.name} 注册失败:`, error);
      throw error;
    }
  }
  
  // 查询所有数据源
  async queryAll(query: string, options?: QueryOptions): Promise<any[]> {
    const results: any[] = [];
    const activeSources = this.getActiveDataSources();
    
    for (const source of activeSources) {
      try {
        const sourceResults = await source.query(query, options);
        results.push(...sourceResults);
      } catch (error) {
        console.error(`数据源 ${source.name} 查询失败:`, error);
        // 继续查询其他数据源
      }
    }
    
    return results;
  }
  
  // 获取活跃数据源
  private getActiveDataSources(): DataSource[] {
    return Array.from(this.dataSources.values())
      .filter(source => this.activeConnections.has(source.name));
  }
  
  // 健康检查
  async healthCheck(): Promise<Record<string, boolean>> {
    const results: Record<string, boolean> = {};
    
    for (const [name, source] of this.dataSources) {
      try {
        results[name] = await source.healthCheck();
      } catch (error) {
        results[name] = false;
      }
    }
    
    return results;
  }
  
  // 断开所有连接
  async disconnectAll(): Promise<void> {
    for (const source of this.dataSources.values()) {
      try {
        await source.disconnect();
      } catch (error) {
        console.error(`数据源 ${source.name} 断开失败:`, error);
      }
    }
    this.activeConnections.clear();
  }
}
```

```typescript
// src/drawing-review/algorithms/ml.algorithm.ts
import { SuggestionAlgorithm, AlgorithmInput, AlgorithmOutput, AlgorithmConfig } from './base.algorithm';

export class MLAlgorithm implements SuggestionAlgorithm {
  name = "machine-learning";
  version = "1.0.0";
  description = "基于机器学习的联想算法";
  
  private config: AlgorithmConfig;
  private model: any; // 机器学习模型
  
  async execute(input: AlgorithmInput): Promise<AlgorithmOutput> {
    try {
      // 特征提取
      const features = this.extractFeatures(input);
      
      // 模型预测
      const predictions = await this.model.predict(features);
      
      // 结果转换
      const suggestions = this.convertToSuggestions(predictions, input);
      
      return {
        suggestions,
        confidence: this.calculateConfidence(predictions),
        reasoning: this.generateReasoning(predictions, input),
        metadata: {
          algorithm: this.name,
          version: this.version,
          timestamp: new Date().toISOString()
        }
      };
    } catch (error) {
      console.error('机器学习算法执行失败:', error);
      return {
        suggestions: [],
        confidence: 0,
        reasoning: "算法执行失败",
        metadata: { error: error.message }
      };
    }
  }
  
  configure(config: AlgorithmConfig): void {
    this.config = config;
    // 加载机器学习模型
    this.loadModel();
  }
  
  validate(input: AlgorithmInput): boolean {
    // 验证输入数据
    return input.query && input.query.length >= 2;
  }
  
  private extractFeatures(input: AlgorithmInput): number[] {
    // 特征提取逻辑
    const features: number[] = [];
    
    // 查询特征
    features.push(input.query.length);
    features.push(input.query.charCodeAt(0) || 0);
    
    // 上下文特征
    if (input.context.projectId) {
      features.push(input.context.projectId.length);
    }
    
    // 用户偏好特征
    if (input.userPreferences) {
      features.push(input.userPreferences.defaultProjectId?.length || 0);
    }
    
    return features;
  }
  
  private async loadModel(): Promise<void> {
    // 加载预训练模型
    // 这里可以集成 TensorFlow.js 或其他机器学习框架
    console.log('加载机器学习模型...');
  }
  
  private convertToSuggestions(predictions: any[], input: AlgorithmInput): SuggestionResult[] {
    // 将模型预测结果转换为联想建议
    return predictions.map(pred => ({
      type: this.predictType(pred),
      value: pred.value,
      label: pred.label,
      count: pred.confidence * 100,
      recent: pred.recency > 0.5
    }));
  }
  
  private calculateConfidence(predictions: any[]): number {
    // 计算整体置信度
    if (predictions.length === 0) return 0;
    
    const avgConfidence = predictions.reduce((sum, pred) => sum + pred.confidence, 0) / predictions.length;
    return Math.min(avgConfidence, 1.0);
  }
  
  private generateReasoning(predictions: any[], input: AlgorithmInput): string {
    // 生成推荐理由
    const reasons: string[] = [];
    
    if (predictions.length > 0) {
      reasons.push(`基于 ${predictions.length} 个预测结果`);
    }
    
    if (input.userPreferences) {
      reasons.push('考虑用户历史偏好');
    }
    
    if (input.historicalData && input.historicalData.length > 0) {
      reasons.push('参考历史数据模式');
    }
    
    return reasons.join('，') || '基于算法分析';
  }
  
  private predictType(prediction: any): string {
    // 预测联想类型
    return prediction.type || 'unknown';
  }
```

### 3. 算法管理器

**算法管理器**:
```typescript
// src/drawing-review/algorithms/algorithm.manager.ts
import { SuggestionAlgorithm, AlgorithmInput, AlgorithmOutput, AlgorithmConfig } from './base.algorithm';

export class AlgorithmManager {
  private algorithms: Map<string, SuggestionAlgorithm> = new Map();
  private configs: Map<string, AlgorithmConfig> = new Map();
  
  // 注册算法
  registerAlgorithm(algorithm: SuggestionAlgorithm, config: AlgorithmConfig): void {
    algorithm.configure(config);
    this.algorithms.set(algorithm.name, algorithm);
    this.configs.set(algorithm.name, config);
    console.log(`算法 ${algorithm.name} 注册成功`);
  }
  
  // 执行所有算法
  async executeAlgorithms(input: AlgorithmInput): Promise<AlgorithmOutput[]> {
    const results: AlgorithmOutput[] = [];
    const enabledAlgorithms = this.getEnabledAlgorithms();
    
    for (const algorithm of enabledAlgorithms) {
      try {
        if (!algorithm.validate(input)) {
          console.warn(`算法 ${algorithm.name} 输入验证失败`);
          continue;
        }
        
        const result = await algorithm.execute(input);
        results.push(result);
      } catch (error) {
        console.error(`算法 ${algorithm.name} 执行失败:`, error);
      }
    }
    
    return results;
  }
  
  // 获取启用的算法
  private getEnabledAlgorithms(): SuggestionAlgorithm[] {
    return Array.from(this.algorithms.values())
      .filter(algorithm => {
        const config = this.configs.get(algorithm.name);
        return config?.enabled;
      })
      .sort((a, b) => {
        const configA = this.configs.get(a.name);
        const configB = this.configs.get(b.name);
        return (configB?.weight || 0) - (configA?.weight || 0);
      });
  }
  
  // 融合算法结果
  async fuseResults(results: AlgorithmOutput[]): Promise<SuggestionResult[]> {
    const fusedSuggestions: SuggestionResult[] = [];
    const suggestionMap = new Map<string, SuggestionResult>();
    
    // 按权重融合结果
    for (const result of results) {
      for (const suggestion of result.suggestions) {
        const key = `${suggestion.type}:${suggestion.value}`;
        
        if (suggestionMap.has(key)) {
          // 合并重复建议
          const existing = suggestionMap.get(key)!;
          existing.count = (existing.count || 0) + (suggestion.count || 0);
        } else {
          suggestionMap.set(key, { ...suggestion });
        }
      }
    }
    
    // 转换为数组并排序
    return Array.from(suggestionMap.values())
      .sort((a, b) => (b.count || 0) - (a.count || 0));
  }
}
```

```typescript
// src/drawing-review/components/CustomSuggestionBox.ts
export class CustomSuggestionBox {
  private container: HTMLElement;
  private input: HTMLInputElement;
  private suggestionsList: HTMLElement;
  private config: SuggestionBoxConfig;
  
  constructor(containerId: string, config: SuggestionBoxConfig) {

  }

  }
  
  private createInput(): void {
    this.input = document.createElement('input');
    this.input.type = 'text';
    this.input.placeholder = this.config.placeholder || '请输入...';
    this.input.className = this.config.inputClassName || 'suggestion-input';

    this.container.appendChild(this.suggestionsList);

    this.input.addEventListener('input', (e) => {
      this.handleInput((e.target as HTMLInputElement).value);
    });

    this.input.addEventListener('focus', () => {

    });
    
    // 失焦事件

      // 延迟隐藏，允许点击建议项

      this.handleKeydown(e);
    });

      this.showError('获取建议失败');
    }
  }
  
  private async fetchSuggestions(query: string): Promise<SuggestionResult[]> {
    const response = await fetch(`/api/drawing-review/suggestions?q=${encodeURIComponent(query)}`);
    const data = await response.json();
    return data.suggestions || [];
  }
  
  private displaySuggestions(suggestions: SuggestionResult[]): void {
    this.suggestionsList.innerHTML = '';
    
    suggestions.forEach((suggestion, index) => {
      const item = this.createSuggestionItem(suggestion, index);
      this.suggestionsList.appendChild(item);
    });
    
    this.showSuggestions();
  }
  
  private createSuggestionItem(suggestion: SuggestionResult, index: number): HTMLElement {
    const item = document.createElement('div');
    item.className = 'suggestion-item';
    item.dataset.index = index.toString();
    
    item.innerHTML = `
      <div class="suggestion-label">${suggestion.label}</div>
      ${suggestion.count ? `<div class="suggestion-count">使用 ${suggestion.count} 次</div>` : ''}
    `;
    
    item.addEventListener('click', () => {
      this.selectSuggestion(suggestion);
    });
    
    return item;
  }
  
  private selectSuggestion(suggestion: SuggestionResult): void {
    this.input.value = suggestion.value;
    this.hideSuggestions();
    
    // 触发选择事件
    this.container.dispatchEvent(new CustomEvent('suggestionSelected', {
      detail: suggestion
    }));
  }
  
  private handleKeydown(e: KeyboardEvent): void {
    const items = this.suggestionsList.querySelectorAll('.suggestion-item');
    const currentIndex = this.getCurrentIndex();
    
    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault();
        this.setCurrentIndex(Math.min(currentIndex + 1, items.length - 1));
        break;
      case 'ArrowUp':
        e.preventDefault();
        this.setCurrentIndex(Math.max(currentIndex - 1, 0));
        break;
      case 'Enter':
        e.preventDefault();
        if (currentIndex >= 0) {
          const selectedItem = items[currentIndex] as HTMLElement;
          selectedItem.click();
        }
        break;
      case 'Escape':
        this.hideSuggestions();
        break;
    }
  }
  
  private getCurrentIndex(): number {
    const selected = this.suggestionsList.querySelector('.suggestion-item.selected');
    return selected ? parseInt(selected.dataset.index || '0') : -1;
  }
  
  private setCurrentIndex(index: number): void {
    const items = this.suggestionsList.querySelectorAll('.suggestion-item');
    items.forEach((item, i) => {
      item.classList.toggle('selected', i === index);
    });
  }
  
  private showSuggestions(): void {
    this.suggestionsList.style.display = 'block';
  }
  
  private hideSuggestions(): void {
    this.suggestionsList.style.display = 'none';
  }
  
  private showError(message: string): void {
    this.suggestionsList.innerHTML = `<div class="suggestion-error">${message}</div>`;
    this.showSuggestions();
  }
}
```

### 1. 缓存优化

**多级缓存实现**:
```typescript
// src/drawing-review/cache/cache.manager.ts
export class CacheManager {
  private l1Cache: Map<string, CacheItem> = new Map(); // 内存缓存
  private l2Cache: any; // KV 缓存
  
  constructor(kvNamespace: any) {
    this.l2Cache = kvNamespace;
  }
  
  // 获取缓存
  async get(key: string): Promise<any> {
    // L1 缓存
    if (this.l1Cache.has(key)) {
      const item = this.l1Cache.get(key)!;
      if (!this.isExpired(item)) {
        return item.value;
      }
      this.l1Cache.delete(key);
    }
    
    // L2 缓存
    try {
      const item = await this.l2Cache.get(key);
      if (item && !this.isExpired(item)) {
        // 回填 L1 缓存
        this.l1Cache.set(key, item);
        return item.value;
      }
    } catch (error) {
      console.error('L2 缓存读取失败:', error);
    }
    
    return null;
  }
  
  // 设置缓存
  async set(key: string, value: any, ttl: number = 300): Promise<void> {
    const item: CacheItem = {
      value,
      timestamp: Date.now(),
      ttl
    };
    
    // L1 缓存
    this.l1Cache.set(key, item);
    
    // L2 缓存
    try {
      await this.l2Cache.put(key, JSON.stringify(item), { expirationTtl: ttl });
    } catch (error) {
      console.error('L2 缓存写入失败:', error);
    }
  }
  
  // 删除缓存
  async delete(key: string): Promise<void> {
    this.l1Cache.delete(key);
    try {
      await this.l2Cache.delete(key);
    } catch (error) {
      console.error('L2 缓存删除失败:', error);
    }
  }
  
  // 清空缓存
  async clear(): Promise<void> {
    this.l1Cache.clear();
    // L2 缓存清空需要根据具体实现
  }
  
  private isExpired(item: CacheItem): boolean {
    return Date.now() - item.timestamp > item.ttl * 1000;
  }
}

// 缓存项接口
interface CacheItem {
  value: any;
  timestamp: number;
  ttl: number;
}
```

```typescript
// src/drawing-review/utils/debounce.ts
export class Debounce {
  private timeouts: Map<string, NodeJS.Timeout> = new Map();
  
  // 防抖执行
  debounce(key: string, func: Function, delay: number): void {
    // 清除之前的定时器
    if (this.timeouts.has(key)) {
      clearTimeout(this.timeouts.get(key)!);
    }
    
    // 设置新的定时器
    const timeout = setTimeout(() => {
      func();
      this.timeouts.delete(key);
    }, delay);
    
    this.timeouts.set(key, timeout);
  }
  
  // 取消防抖
  cancel(key: string): void {
    if (this.timeouts.has(key)) {
      clearTimeout(this.timeouts.get(key)!);
      this.timeouts.delete(key);
    }
  }
  
  // 取消防抖所有
  cancelAll(): void {
    for (const timeout of this.timeouts.values()) {
      clearTimeout(timeout);
    }
    this.timeouts.clear();
  }
}
```
